#übung 1 tage bis zum Ball 
```sh
vim uebung_1.sh
#!/bin/bash

htl_ball="2027-01-15"

ziel=$(date -d "$htl_ball" +%s)
jetzt=$(date +%s)
tage=$(( (ziel-jetzt)/ 86400))
echo" es sind $tage Tage bis zum htl ball"
```
chmod macht es ausführbar und ./ führt es aus 
```sh
chmod +x uebung_1.sh

./uebung_1.sh
```

```sh

#!/bin/bash

eingabe_datei=$1

heute=$(date +%F)

neue_datei="${heute}_${eingabe_datei}"

cp "$eingabe_datei" "$neue_datei"

echo "Kopie erfolgreich erstellt: $neue_datei"
```
